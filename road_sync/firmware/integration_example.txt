Implementasi road_uart_send wajib ditambahkan ke proyek, contoh transmit IT:

static uint8_t esp_tx[180];
static volatile bool esp_tx_busy;
bool road_uart_send(const char *line) {
    size_t n = strlen(line);
    if (esp_tx_busy || n >= sizeof esp_tx) return false;
    memcpy(esp_tx, line, n);
    esp_tx_busy = true;
    if (HAL_UART_Transmit_IT(&huart_esp, esp_tx, (uint16_t)n) != HAL_OK) {
        esp_tx_busy = false;
        return false;
    }
    return true;
}
void HAL_UART_TxCpltCallback(UART_HandleTypeDef *h) {
    if (h == &huart_esp) esp_tx_busy = false;
}
// Gabungkan dengan callback yang sudah ada; jangan definisikan dua kali.
// Sertakan string.h. huart_esp merupakan placeholder, ganti dengan handle asli.
// RX: gunakan HAL_UART_Receive_IT atau DMA + ring buffer.
// ISR hanya memasukkan byte ke buffer; FatFs dan road_receive berjalan di main.
// Tangani UART error/overrun dan aktifkan kembali receive.
